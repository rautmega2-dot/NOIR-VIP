function renderBuyConfirm() {
  const p = planData.find(x => x.productId === pendingProductId);
  if (!p) return;
  const isVip = p.projectType === 'vip';
  const price = Number(p.price) || 0;
  const daily = Number(p.dailyProfit) || 0;
  const days = Number(p.validityDays || p.totalDays) || 0;
  const bal = walletBalanceNum;
  const lim = limitInfo(p);
  const remaining = Math.max(0, lim.limit - lim.active);
  const byBalance = bal === null || price <= 0 ? remaining : Math.floor(bal / price);
  let maxQty = remaining <= 0 ? 1 : Math.max(1, Math.min(remaining, byBalance, 20));
  if (buyQty > maxQty) buyQty = maxQty;
  if (buyQty < 1) buyQty = 1;
  const qty = buyQty;
  const totalPay = Math.round(price * qty * 1e6) / 1e6;
  const totalDaily = Math.round(daily * qty * 1e6) / 1e6;
  const totalReturn = Math.round(daily * days * qty * 1e6) / 1e6;
  const enough = bal !== null && bal >= totalPay;

  document.getElementById('bc-type').textContent = isVip ? 'VIP Project' : 'NOIR Project';
  document.getElementById('bc-name').textContent = p.name || 'Project';
  document.getElementById('bc-details').innerHTML =
    bcLine(qty > 1 ? 'Price (each)' : 'Price', money(price), qty === 1) +
    bcLine('Daily Profit', money(totalDaily)) +
    bcLine('Duration', days + ' days') +
    bcLine('Total Return', money(totalReturn)) +
    bcLine('Payout Time', 'Daily 7–9 PM') +
    bcLine('Your Active', lim.active + ' / ' + lim.limit) +
    bcLine('Your Balance', bal === null ? 'Loading…' : money(bal)) +
    (qty > 1 ? bcLine('Total Price', money(totalPay), true) : '');

  const qtyRow = document.getElementById('bc-qty-row');
  const hint = document.getElementById('bc-hint');
  const showQty = remaining >= 2;
  qtyRow.style.display = showQty ? 'flex' : 'none';
  document.getElementById('bc-qty').textContent = qty;
  document.getElementById('bc-minus').disabled = buying || qty <= 1;
  document.getElementById('bc-plus').disabled = buying || qty >= maxQty;
  if (showQty && qty >= maxQty && bal !== null && bal >= price) {
    hint.textContent = byBalance < remaining ? 'Your balance allows up to ' + maxQty : 'Limit: you can buy up to ' + maxQty + ' now';
    hint.style.display = 'block';
  } else { hint.style.display = 'none'; }

  const warn = document.getElementById('bc-warn');
  const btn = document.getElementById('bc-confirm');
  let warnText = '';
  if (buyError) warnText = buyError;
  else if (lim.reached) warnText = 'Limit reached. You already have ' + lim.active + ' of ' + lim.limit + ' active for this project.' + (lim.nextDays > 0 ? ' A slot opens in about ' + lim.nextDays + ' day' + (lim.nextDays === 1 ? '' : 's') + '.' : '');
  else if (bal !== null && !enough) warnText = 'Insufficient balance. Please deposit ' + money(totalPay - bal) + ' more to buy this project.';
  if (warnText) { warn.textContent = warnText; warn.classList.remove('hidden'); } else { warn.classList.add('hidden'); }
  btn.classList.toggle('vip-buy', isVip);
  btn.disabled = buying || bal === null || !enough || lim.reached;
  btn.textContent = buying ? 'Processing...' : ('Buy & Pay ' + money(totalPay));
}

async function openBuyConfirm(productId) {
  const token = typeof getSessionToken === 'function' ? getSessionToken() : null;
  if (!token) { toast('Please sign in to purchase'); if (typeof openAuth === 'function') openAuth(); return; }
  const chosen = planData.find(x => x.productId === productId);
  if (!chosen) return;
  if (chosen.projectType === 'vip' && !isVipLive(chosen)) { toast('This VIP sale has ended'); renderPlans(); return; }
  pendingProductId = productId;
  buyError = '';
  buyQty = 1;
  walletBalanceNum = null;
  renderBuyConfirm();
  document.getElementById('buy-modal').classList.add('open');
  try { await loadWalletBalance(); } catch (e) {}
  if (pendingProductId === productId) renderBuyConfirm();
}

  if (buying) return;
  pendingProductId = null;
  document.getElementById('buy-modal').classList.remove('open');
}

async function confirmBuy() {
  if (buying || !pendingProductId) return;
  const token = typeof getSessionToken === 'function' ? getSessionToken() : null;
  if (!token) { toast('Please sign in to purchase'); return; }
  const productId = pendingProductId;
  buyError = '';
  buying = true;
  renderBuyConfirm();
  try {
    const res = await fetch('/api/project', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
      body: JSON.stringify({ productId, quantity: buyQty })
    });
    const data = await res.json();
    if (!res.ok || !data.success) throw new Error(data.message || 'Purchase failed');

    buying = false;
    pendingProductId = null;
    document.getElementById('buy-modal').classList.remove('open');
    toast(data.message || 'Project purchased successfully!');
    await fetchLiveProjects();
    if (typeof loadWalletBalance === 'function') await loadWalletBalance();
    go('product');
    setTab('mine', document.querySelector('[data-tab=mine]'));
  } catch (err) {
    buying = false;
    buyError = err.message || 'Purchase failed';
    try { await loadWalletBalance(); } catch (e) {}
    try { await fetchLiveProjects(); } catch (e) {}
    renderBuyConfirm();
  }
}

async function submitPasswordChange() {

  const nw = document.getElementById('pw-new').value;
  const cf = document.getElementById('pw-confirm').value;
  const st = document.getElementById('pw-status');

  const fail = msg => { st.style.color = '#ffb4b4'; st.textContent = msg; };
  if (!cur) return fail('Enter your current password.');
  if (nw.length < 8 || nw.length > 128) return fail('New password must be 8–128 characters.');
  if (nw === cur) return fail('New password must be different from the current one.');
  if (nw !== cf) return fail('New passwords do not match.');
  pwBusy = true; btn.disabled = true; btn.textContent = 'Updating...'; st.textContent = '';
  try {
    const res = await fetch('/api/auth', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + getSessionToken() },
      body: JSON.stringify({ action: 'changePassword', currentPassword: cur, newPassword: nw })
    });
    const data = await res.json().catch(() => ({}));
    if (!res.ok || !data.success) throw new Error(data.error || 'Could not update password. Please try again.');
    pwBusy = false;
    ['pw-current', 'pw-new', 'pw-confirm'].forEach(id => { document.getElementById(id).value = ''; });
    document.getElementById('password-modal').classList.remove('open');
    toast('Password updated');
  } catch (err) {
    fail(err.message || 'Could not update password.');
  } finally {
    pwBusy = false; btn.disabled = false; btn.textContent = 'Update Password';
  }

async function applyPendingReferral(token, noirId) {
  const referrerNoirId = pendingReferral();

    const result = await response.json();

  if(input&&input.value.trim()!==''&&(!Number.isFinite(Number(input.value))||Number(input.value)<dMin||Number(input.value)>dMax)){showMessage('Enter an amount from '+payFmt(dMin)+' to '+payFmt(dMax)+' USDT.',true);return;}

  const token=getSessionToken();
  if(!token){showMessage('Please log in first.',true);return;}
  const originalText=btn.innerHTML;
  btn.disabled=true;btn.textContent='Creating invoice...';
  showMessage('Creating your secure payment invoice...');
  try{
    const response=await fetch('/api/deposit',{method:'POST',headers:{'Content-Type':'application/json',Authorization:`Bearer ${token}`},body:JSON.stringify({amount:noirDepositAmount}),cache:'no-store'});
    const result=await response.json().catch(()=>({}));
    if(!response.ok||!result.success||!result.paymentUrl){showMessage(result.message||'Could not start payment. Please try again.',true);return;}
    showMessage('Opening secure payment page...');
    window.location.href=result.paymentUrl;
  }catch(error){
    console.error('Deposit request failed',error);
    showMessage('Network error. Please try again.',true);
  }finally{
    btn.disabled=false;btn.innerHTML=originalText;
  }
}

async function submitWithdraw(){
  const amountEl=document.getElementById('withdraw-amount');
  const addressEl=document.getElementById('withdraw-address');
  const feedback=document.getElementById('withdraw-feedback');
  const btn=document.getElementById('withdraw-submit-btn');
  const show=(m,err=false)=>{if(feedback){feedback.textContent=m;feedback.style.display='block';feedback.style.borderColor=err?'#a84b4b':'#ffffff22';feedback.style.color=err?'#ffb6b6':'#ddd';}};
  const amount=Number(amountEl&&amountEl.value);
  const address=String(addressEl&&addressEl.value||'').trim();
  const wMin=noirPay.minWithdrawal,wMax=noirPay.maxWithdrawal;
  if(!Number.isFinite(amount)||amount<wMin||amount>wMax){show('Withdrawal amount must be between '+payFmt(wMin)+' and '+payFmt(wMax)+' USDT.',true);return;}
  if(!address){show('Enter your USDT wallet address.',true);return;}
  if(!/^0x[a-fA-F0-9]{40}$/.test(address)){show('Enter a valid wallet address starting with 0x.',true);return;}
  const token=getSessionToken();
  if(!token){show('Please log in first.',true);return;}
  const original=btn.innerHTML;btn.disabled=true;btn.textContent='Processing...';
  show('Processing your withdrawal...');
  try{
    const response=await fetch('/api/withdraw',{method:'POST',headers:{'Content-Type':'application/json',Authorization:`Bearer ${token}`},body:JSON.stringify({amount,address,network:noirWithdrawNetwork}),cache:'no-store'});
    const result=await response.json().catch(()=>({}));
    if(!response.ok||!result.success){
      show(result.message||'Withdrawal failed. Please try again.',true);
      if(result.code==='project_required'&&feedback){
        const go=document.createElement('button');go.type='button';go.textContent='Browse Projects →';
        go.style.cssText='display:block;margin-top:10px;background:none;border:0;padding:0;color:#fff;font-size:12px;font-weight:700;text-decoration:underline;cursor:pointer';
        go.addEventListener('click',()=>{closeWithdrawModal();openAvailableProjects();});
        feedback.appendChild(go);
      }
      return;
    }
    const got=(result.netAmount!==undefined?result.netAmount:result.amount);
    if(result.status==='sent')show('Withdrawal of '+got+' USDT has been sent.');
    else if(result.status==='processing')show('Withdrawal of '+got+' USDT is being processed. Check History for the final status.');
    else if(result.status==='sandbox_simulated')show('Test mode: withdrawal of '+got+' USDT was simulated. No real USDT was sent.');
    else show('Your withdrawal is under review.');
    amountEl.value='';addressEl.value='';updateWithdrawCharge();
    await loadWalletBalance();
    const bal=document.getElementById('wallet-balance');const av=document.getElementById('withdraw-available');
    if(av&&bal)av.textContent=bal.textContent;
  }catch(error){
    console.error('Withdraw request failed',error);
    show('Network error. Please try again.',true);
  }finally{btn.disabled=false;btn.innerHTML=original;}
}

  deposit:{pending:'Waiting for payment',paid:'Completed',review_required:'Under review',invoice_failed:'Failed',expired:'Expired'},

async function loadHistory(){
  const box=document.getElementById('history-list');if(!box)return;
  const note=(text)=>{box.textContent='';const p=document.createElement('p');p.style.cssText='color:#929292;font-size:12px;text-align:center;padding:34px 0;line-height:1.6';p.textContent=text;box.appendChild(p);};
  note('Loading...');
  try{
    const response=await fetch('/api/history',{headers:{Authorization:`Bearer ${getSessionToken()}`},cache:'no-store'});
    const raw=await response.text();
    let result=null;try{result=JSON.parse(raw);}catch(e){}
    if(!response.ok||!result||!result.success){
      note('Could not load history (error '+response.status+'). '+((result&&result.message)||'Please try again.'));
      return;
    }
    if(!result.items.length){note('No wallet activity yet.');return;}
    box.textContent='';
    result.items.forEach(item=>{
      const kind=item.kind||(item.type==='withdrawal'?'withdrawal':'deposit');
      const TITLES={deposit:'Deposit',withdrawal:'Withdrawal',purchase:'Project Purchase',commission:'Referral Commission',daily_profit:'Daily Profit',salary:'Monthly Salary'};
      const COLORS={deposit:'#c6f6d5',daily_profit:'#c6f6d5',commission:'#ffe9a3',salary:'#ffe9a3',purchase:'#f4f4f4',withdrawal:'#f4f4f4'};
      const isOut=kind==='withdrawal'||kind==='purchase';
      const isLedger=kind==='purchase'||kind==='commission'||kind==='daily_profit'||kind==='salary';
      const row=document.createElement('div');
      row.style.cssText='display:flex;justify-content:space-between;gap:12px;padding:14px 14px;margin-bottom:10px;border:1px solid #ffffff14;border-radius:16px;background:#ffffff06';
      const left=document.createElement('div');
      const title=document.createElement('div');title.style.cssText='font-size:14px;color:#f4f4f4';title.textContent=TITLES[kind]||'Wallet';
      const sub=document.createElement('div');sub.style.cssText='font-size:11px;color:#929292;margin-top:4px;line-height:1.5';
      const when=item.createdAt?new Date(item.createdAt).toLocaleString():'';
      let extra='';
      if(kind==='withdrawal')extra=(item.network?' · '+item.network:'')+(item.address?' · '+item.address:'');
      else if(kind==='purchase')extra=(item.projectName?' · '+item.projectName:'')+(item.quantity>1?' × '+item.quantity:'');
      else if(kind==='commission'&&item.level)extra=' · Level '+item.level;
      sub.textContent=when+extra;
      left.appendChild(title);left.appendChild(sub);
      const right=document.createElement('div');right.style.cssText='text-align:right;white-space:nowrap';
      const amt=document.createElement('div');amt.style.cssText='font-size:14px;color:'+(COLORS[kind]||'#f4f4f4');
      amt.textContent=(isOut?'-':'+')+Math.abs(Number(item.amount)||0).toLocaleString('en-US',{maximumFractionDigits:2})+' '+(item.currency||'USDT');
      const st=document.createElement('div');st.style.cssText='font-size:11px;color:#929292;margin-top:4px';
      st.textContent=(isLedger&&item.status==='completed')?'Completed':((HISTORY_STATUS[item.type]&&HISTORY_STATUS[item.type][item.status])||item.status);
      right.appendChild(amt);right.appendChild(st);
      row.appendChild(left);row.appendChild(right);box.appendChild(row);
    });
  }catch(error){
    console.error('History load failed',error);
    note('Could not load history. Check your connection and try again.');
  }
}
